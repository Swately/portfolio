% ── External: the game ───────────────────────────────────────────────
\node[ext] (game) {\textbf{Game}\\(external process)};

% ── Thread C — capture, on the iGPU ──────────────────────────────────
\node[work, fill=figFill, right=18mm of game] (cap)
  {\textbf{Capture}\,(thread C)\\DDA / WGC grab\\$\to$ convert + pack};
\node[badge, below=5mm of cap] (igpu) {iGPU};
\node[rolelab, below=1.5mm of igpu] (igpurole) {zero-copy convert+pack\\(no VRAM; system RAM)};

% ── Thread F — optical flow, on the 1080 Ti ──────────────────────────
\node[work, fill=figFill, right=20mm of cap] (flow)
  {\textbf{Optical flow}\,(thread F)\\MV + SAD pyramid\\+ global model (gme)};
\node[badge, below=5mm of flow] (g1080) {GTX 1080 Ti};
\node[rolelab, below=1.5mm of g1080] (g1080role) {optical-flow workhorse\\(capped compute)};

% ── Thread P — warp + present, on the 4090 ───────────────────────────
\node[work, fill=figFill, right=20mm of flow, text width=3.6cm] (pres)
  {\textbf{Warp\,+\,blend\,+\,present}\\(thread P)\\backward-warp to $t$,\\quality, pacing};
\node[badge, below=5mm of pres] (g4090) {RTX 4090};
\node[rolelab, below=1.5mm of g4090] (g4090role) {warp + present\\(owns the display)};

% ── Display ──────────────────────────────────────────────────────────
\node[ext, right=18mm of pres, fill=figFill] (disp) {\textbf{Display}\\(e.g. 240\,Hz)};

% ── Data flow (FgContext: SPSC rings + sequence counters) ────────────
\draw[->, very thick, figMuted] (game) -- (cap) node[ringlab]{frames};
\draw[->, very thick, figMuted] (cap)  -- (flow) node[ringlab]{\texttt{c\_seq}\\(real pairs)};
\draw[->, very thick, figMuted] (flow) -- (pres) node[ringlab]{\texttt{f\_seq}\\(MV/SAD + reals)};
\draw[->, very thick] (pres) -- (disp) node[ringlab, text=figMuted]{present\\(paced)};
